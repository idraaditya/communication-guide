# 802.11s (HWMP) on ESP32-S3 - Complete Setup Guide

## Prerequisites

### Hardware
- **ESP32-S3 DevKitC-1** (or any ESP32-S3 with WiFi)
- **ESP32-C6** (for WiFi 6 support)
- **Multiple devices** (minimum 3 for mesh)

### Software
- **ESP-IDF v5.2+** (for 802.11s support)
- **Linux host** for testing (optional)

### 2. Basic Mesh Code

```c
// main/mesh_basic.c
#include <string.h>
#include "esp_wifi.h"
#include "esp_netif.h"
#include "esp_event.h"
#include "esp_log.h"
#include "nvs_flash.h"
#include "esp_mesh.h"

static const char *TAG = "mesh_basic";

#define MESH_SSID "esp32-mesh"
#define MESH_PASS "meshpassword"
#define MESH_CHANNEL 6
#define MESH_MAX_LAYER 10

static void mesh_event_handler(void *arg, esp_event_base_t event_base,
                               int32_t event_id, void *event_data) {
    switch (event_id) {
        case MESH_EVENT_STARTED:
            ESP_LOGI(TAG, "Mesh started");
            break;
        case MESH_EVENT_PARENT_CONNECTED:
            mesh_event_connected_t *conn = (mesh_event_connected_t *)event_data;
            ESP_LOGI(TAG, "Parent connected: " MACSTR, MAC2STR(conn->connected.bssid));
            break;
        case MESH_EVENT_PARENT_DISCONNECTED:
            ESP_LOGI(TAG, "Parent disconnected");
            break;
        case MESH_EVENT_CHILD_CONNECTED:
            mesh_event_child_connected_t *child = (mesh_event_child_connected_t *)event_data;
            ESP_LOGI(TAG, "Child connected: " MACSTR, MAC2STR(child->mac));
            break;
        case MESH_EVENT_CHILD_DISCONNECTED:
            mesh_event_child_disconnected_t *child = (mesh_event_child_disconnected_t *)event_data;
            ESP_LOGI(TAG, "Child disconnected: " MACSTR, MAC2STR(child->mac));
            break;
        case MESH_EVENT_ROUTING_TABLE_ADD:
            ESP_LOGI(TAG, "Route added");
            break;
        case MESH_EVENT_ROUTING_TABLE_REMOVE:
            ESP_LOGI(TAG, "Route removed");
            break;
        case MESH_EVENT_ROOT_GOT_IP:
            mesh_event_root_got_ip_t *ip = (mesh_event_root_got_ip_t *)event_data;
            ESP_LOGI(TAG, "Root got IP: " IPSTR, IP2STR(&ip->ip_info.ip));
            break;
        default:
            break;
    }
}

void app_main(void) {
    // Initialize NVS
    ESP_ERROR_CHECK(nvs_flash_init());
    
    // Initialize TCP/IP stack
    ESP_ERROR_CHECK(esp_netif_init());
    ESP_ERROR_CHECK(esp_event_loop_create_default());

    // Initialize WiFi
    wifi_init_config_t cfg = WIFI_INIT_CONFIG_DEFAULT();
    ESP_ERROR_CHECK(esp_wifi_init(&cfg));
    
    // Register mesh event handler
    ESP_ERROR_CHECK(esp_event_handler_register(MESH_EVENT, ESP_EVENT_ANY_ID, 
                                               &mesh_event_handler, NULL));
    
    // Configure mesh
    ESP_ERROR_CHECK(esp_mesh_init());
    
    // Set mesh ID
    mesh_cfg_t cfg = {
        .mesh_id = {0},
        .channel = 6,
        .router = {
            .ssid = "router_ssid",
            .password = "router_pass",
        },
        .mesh_ap = {
            .password = "meshpass",
            .max_connection = 10,
        },
    };
    memcpy(cfg.mesh_id, "esp32-mesh", 10);
    memcpy(cfg.mesh_ap.password, "meshpass", 8);
    
    ESP_ERROR_CHECK(esp_mesh_set_config(&cfg));
    
    // Start mesh
    ESP_ERROR_CHECK(esp_mesh_start());
    
    // Set root node (optional - one node as root)
    // esp_mesh_set_self_organized(false);
    // esp_mesh_set_root(true, true);
    
    ESP_LOGI(TAG, "Mesh started");
    
    // Main loop
    while (1) {
        vTaskDelay(pdMS_TO_TICKS(10000));
        
        // Print mesh info
        mesh_assoc_t assoc;
        esp_mesh_get_assoc(&assoc);
        ESP_LOGI(TAG, "Layer: %d, Parent: " MACSTR ", Children: %d", 
                 assoc.layer, MAC2STR(assoc.parent_bssid), assoc.child_num);
        
        // Print routing table
        int size = esp_mesh_get_routing_table_size();
        mesh_addr_t *route_table = malloc(size * sizeof(mesh_addr_t));
        esp_mesh_get_routing_table(route_table, &size, size * sizeof(mesh_addr_t));
        ESP_LOGI(TAG, "Routing table size: %d", size);
        free(route_table);
    }
}
```

## Advanced Configuration

### Root Node (Gateway)

```c
// Configure one node as root/gateway
void setup_root_node(void) {
    // Disable self-organization
    esp_mesh_set_self_organized(false);
    
    // Set as root
    esp_mesh_set_root(true, true);
    
    // Set root IP (static)
    esp_netif_ip_info_t ip_info;
    IP4_ADDR(&ip_info.ip, 192, 168, 4, 1);
    IP4_ADDR(&ip_info.gw, 192, 168, 4, 1);
    IP4_ADDR(&ip_info.netmask, 255, 255, 255, 0);
    esp_netif_set_ip_info(esp_netif_get_handle_from_ifkey("WIFI_STA_DEF"), &ip_info);
    
    // Enable DHCP server for children
    esp_netif_dhcps_start(esp_netif_get_handle_from_ifkey("WIFI_STA_DEF"));
}
```

### Non-Root Nodes

```c
// Configure child nodes
void setup_child_node(void) {
```

## Monitoring & Debugging

### Mesh Statistics

```c
void print_mesh_stats(void) {
    // Association info
    mesh_assoc_t assoc;
    esp_mesh_get_assoc(&assoc);
    ESP_LOGI(TAG, "Layer: %d, Parent: " MACSTR ", State: %d", 
             assoc.layer, MAC2STR(assoc.parent_bssid), assoc.state);
    
    // Routing table
    int size = esp_mesh_get_routing_table_size();
    mesh_addr_t *table = malloc(size * sizeof(mesh_addr_t));
    esp_mesh_get_routing_table(table, &size, size * sizeof(mesh_addr_t));
    
    ESP_LOGI(TAG, "Routing table entries: %d", size);
    for (int i = 0; i < size; i++) {
        ESP_LOGI(TAG, "Route %d: " MACSTR " -> " MACSTR, 
                 i, MAC2STR(table[i].addr), MAC2STR(table[i].next_hop));
    }
    free(table);
    
    // Child info
    int child_count = assoc.child_num;
    mesh_addr_t *children = malloc(child_count * sizeof(mesh_addr_t));
    esp_mesh_get_child_info(children, &child_count);
    ESP_LOGI(TAG, "Children: %d", child_count);
    free(children);
    
    // TX/RX stats
    mesh_tx_pending_t pending;
    esp_mesh_get_tx_pending(&pending);
    ESP_LOGI(TAG, "TX pending: %d", pending.pending_num);
    
    // Per-peer stats
    mesh_peer_info_t peer;
    for (int i = 0; i < child_count; i++) {
        esp_mesh_get_peer_info(children[i].addr, &peer);
        ESP_LOGI(TAG, "Peer " MACSTR ": RSSI=%d, Rate=%d", 
                 MAC2STR(peer.mac), peer.rssi, peer.rate);
    }
}
```

### Event Monitoring

```c
// Custom event handler for detailed logging
static void mesh_event_handler(void *arg, esp_event_base_t event_base,
                               int32_t event_id, void *event_data) {
    switch (event_id) {
        case MESH_EVENT_PS_PARENT_DUTY:
            mesh_event_ps_duty_t *duty = (mesh_event_ps_duty_t *)event_data;
            ESP_LOGI(TAG, "PS duty: %d%%", duty->duty);
            break;
        case MESH_EVENT_SCAN_DONE:
            ESP_LOGI(TAG, "Scan done");
            break;
        case MESH_EVENT_NETWORK_STATE:
            mesh_event_network_state_t *state = (mesh_event_network_state_t *)event_data;
            ESP_LOGI(TAG, "Network state: %s", state->is_rootless ? "rootless" : "rooted");
            break;
        case MESH_EVENT_STOP:
            ESP_LOGI(TAG, "Mesh stopped");
            break;
        default:
            break;
    }
}
```

## Performance Tuning

### For Maximum Range

```c
// Channel selection (1, 6, 11 for 2.4 GHz)
#define MESH_CHANNEL 6

// TX power (max)
esp_wifi_set_max_tx_power(78);  // 19.5 dBm

// PHY rates (enable all)
esp_wifi_set_protocol(WIFI_IF_MESH, 
    WIFI_PROTOCOL_11B | WIFI_PROTOCOL_11G | WIFI_PROTOCOL_11N);

// Disable power save
esp_wifi_set_ps(WIFI_PS_NONE);
```

### For Maximum Throughput

```c
// HT40 channel width
wifi_config_t cfg = {0};
esp_wifi_get_config(WIFI_IF_MESH, &cfg);
cfg.mesh.channel = 6;
cfg.mesh.crypto_funcs = &g_wifi_default_crypto_funcs;
esp_wifi_set_config(WIFI_IF_MESH, &cfg);

// HT40 (requires channel 1-7 for 2.4 GHz)
esp_wifi_set_channel(6, WIFI_SECOND_CHAN_ABOVE);

// Disable power save
esp_wifi_set_ps(WIFI_PS_NONE);

// Max TX power
esp_wifi_set_max_tx_power(78);
```

### For Battery Life

```c
// Enable power save
esp_wifi_set_ps(WIFI_PS_MIN_MODEM);

// Reduce TX power

// Not directly configurable in mesh, but reduces beacon overhead
```

## Troubleshooting

| Issue | Cause | Fix |
|-------|-------|-----|
| No mesh formation | Wrong channel/SSID/pass | Verify all nodes match |
| No root elected | All nodes same config | Set one node as root explicitly |
| High latency | Too many hops | Reduce max layer, add root nodes |
| Disconnections | Weak signal | Increase TX power, reduce hops |
| Low throughput | Interference | Change channel, use 5 GHz (C6) |
| Memory issues | Large routing table | Reduce max layer, max peers |
